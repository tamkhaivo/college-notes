CREATE TABLE Student (
    SSN VARCHAR(10) PRIMARY KEY,
    StudentName VARCHAR(100) NOT NULL,
    Address VARCHAR(255),
    Email VARCHAR(100) UNIQUE
);
CREATE TABLE Instructor (
    InstructorID VARCHAR(10) PRIMARY KEY,
    InstructorName VARCHAR(100) NOT NULL,
    Office VARCHAR(50)
);
CREATE TABLE TA (
    SSN VARCHAR(10) PRIMARY KEY,
    Salary DECIMAL(10, 2),
    FOREIGN KEY (SSN) REFERENCES Student(SSN) ON DELETE CASCADE
);
CREATE TABLE Course (
    CourseNo VARCHAR(10) PRIMARY KEY,
    CourseName VARCHAR(100) NOT NULL,
    InstructorID VARCHAR(10),
    NoOfStudents INT DEFAULT 0,
    TASSN VARCHAR(10),
    FOREIGN KEY (InstructorID) REFERENCES Instructor(InstructorID) ON DELETE SET NULL,
    FOREIGN KEY (TASSN) REFERENCES TA(SSN) ON DELETE SET NULL
);
CREATE TABLE OnlineCourse (
    CourseNo VARCHAR(10) PRIMARY KEY,
    URL VARCHAR(255),
    FOREIGN KEY (CourseNo) REFERENCES Course(CourseNo) ON DELETE CASCADE
);
CREATE TABLE InPersonCourse (
    CourseNo VARCHAR(10) PRIMARY KEY,
    ClassTime VARCHAR(50),
    RoomNo VARCHAR(20),
    Building VARCHAR(50),
    FOREIGN KEY (CourseNo) REFERENCES Course(CourseNo) ON DELETE CASCADE
);
CREATE TABLE Enrolled (
    StudentSSN VARCHAR(10),
    CourseNo VARCHAR(10),
    Grade VARCHAR(2),
    PRIMARY KEY (StudentSSN, CourseNo),
    FOREIGN KEY (StudentSSN) REFERENCES Student(SSN) ON DELETE CASCADE,
    FOREIGN KEY (CourseNo) REFERENCES Course(CourseNo) ON DELETE CASCADE
);
