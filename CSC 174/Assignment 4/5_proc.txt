DELIMITER $

CREATE PROCEDURE Get_TA(
    IN in_instructorID VARCHAR(10)
)
BEGIN
    SELECT DISTINCT S.StudentName
    FROM Student S
    JOIN TA T ON S.SSN = T.SSN
    JOIN Course C ON T.SSN = C.TASSN
    WHERE C.InstructorID = in_instructorID;
END;
$

CREATE PROCEDURE GetStudentCourse(
    IN in_CourseNumber VARCHAR(10)
)
BEGIN
    SELECT S.SSN, S.StudentName, S.Address, S.Email
    FROM Student S
    JOIN Enrolled E ON S.SSN = E.StudentSSN
    WHERE E.CourseNo = in_CourseNumber;
END;
$

DELIMITER ;
